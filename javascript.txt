<!DOCTYPE html>
<html>
<head>
    <title>Simple Calculator</title>
</head>

<body>

    <h2>Simple Calculator</h2>

    <!-- Input Fields -->
    First Number: <br>
    <input type="text" id="num1"><br><br>

    Second Number: <br>
    <input type="text" id="num2"><br><br>

    <!-- Buttons -->
    <button onclick="calculate('+')">+</button>
    <button onclick="calculate('-')">-</button>
    <button onclick="calculate('*')">*</button>
    <button onclick="calculate('/')">/</button>

    <script>
        function calculate(op) {
            let a = Number(document.getElementById("num1").value);
            let b = Number(document.getElementById("num2").value);
            let result;

            if (op == '+') {
                result = a + b;
            } 
            else if (op == '-') {
                result = a - b;
            } 
            else if (op == '*') {
                result = a * b;
            } 
            else if (op == '/') {
                if (b == 0) {
                    alert("Cannot divide by zero!");
                    return;
                }
                result = a / b;
            }

            alert("Result = " + result);
        }
    </script>

</body>
</html>

<!DOCTYPE html>
<html>
<head>
    <title>Form Validation</title>
</head>

<body>

    <h2>Student Form</h2>

    <form onsubmit="return validateForm()">

        Roll Number: <br>
        <input type="text" id="roll"><br><br>

        Name: <br>
        <input type="text" id="name"><br><br>

        Date of Birth: <br>
        <input type="date" id="dob"><br><br>

        <input type="submit" value="Submit">

    </form>

    <script>
        // Validate Roll Number (7-digit numeric)
        function validateRoll() {
            let roll = document.getElementById("roll").value;

            if (roll.length != 7 || isNaN(roll)) {
                alert("Roll number must be 7 digits!");
                return false;
            }
            return true;
        }

        // Validate Name (alphabets only)
        function validateName() {
            let name = document.getElementById("name").value;

            let pattern = /^[A-Za-z ]+$/;

            if (!pattern.test(name)) {
                alert("Name must contain only alphabets!");
                return false;
            }
            return true;
        }

        // Validate DOB (non-empty)
        function validateDOB() {
            let dob = document.getElementById("dob").value;

            if (dob == "") {
                alert("Date of Birth cannot be empty!");
                return false;
            }
            return true;
        }

        // Main validation function
        function validateForm() {
            if (!validateRoll()) return false;
            if (!validateName()) return false;
            if (!validateDOB()) return false;

            alert("Form submitted successfully!");
            return true;
        }
    </script>

</body>
</html>
