<!DOCTYPE html>
<html>
<head>
    <title>Semantic Tags Example</title>
</head>

<body>

    <!-- Header -->
    <header style="background-color: lightblue; padding: 10px;">
        <h1>My Website</h1>
        <p>Welcome to my webpage</p>
    </header>

    <!-- Navigation -->
    <nav style="background-color: lightgray; padding: 10px;">
        <a href="#">Home</a> |
        <a href="#">About</a> |
        <a href="#">Contact</a>
    </nav>

    <!-- Main Content -->
    <main>

        <!-- Section -->
        <section>
            <h2>About HTML</h2>
            <p>HTML is used to create webpages.</p>
        </section>

        <!-- Another Section -->
        <section>
            <h2>More Info</h2>

            <!-- Details and Summary -->
            <details>
                <summary>Click to learn more</summary>
                <p>HTML stands for HyperText Markup Language.</p>
            </details>
        </section>

        <!-- Figure -->
        <figure>
            <img src="sample.jpg" width="200">
            <figcaption>This is a sample image</figcaption>
        </figure>

    </main>

    <!-- Footer -->
    <footer style="background-color: lightblue; padding: 10px; text-align: center;">
        <p>© 2026 My Website</p>
    </footer>

</body>
</html>
